cmake_minimum_required(VERSION 3.22.1)
project(lens_jvmti LANGUAGES CXX)

add_library(lens_jvmti SHARED lens_agent.cpp)
set(JVMTI_INCLUDE_DIR "$ENV{JAVA_HOME}/include")
if(NOT EXISTS "${JVMTI_INCLUDE_DIR}/jvmti.h")
    message(FATAL_ERROR "JAVA_HOME must point to an OpenJDK installation containing include/jvmti.h")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/jvmti-include")
configure_file("${JVMTI_INCLUDE_DIR}/jvmti.h" "${CMAKE_CURRENT_BINARY_DIR}/jvmti-include/jvmti.h" COPYONLY)
target_include_directories(lens_jvmti PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/jvmti-include")
target_compile_features(lens_jvmti PRIVATE cxx_std_20)
target_compile_options(lens_jvmti PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_libraries(lens_jvmti PRIVATE log)
